import os
import streamlit as st
import google.generativeai as genai

st.set_page_config(
    page_title="Қазақ Дәстүрлері Комиксте",
    page_icon="🎨",
    layout="centered"
)

api_key = os.environ.get("GEMINI_API_KEY")

if not api_key:
    st.error("⚠️ Gemini API ключы табылмады! Replit Secrets бөліміне 'GEMINI_API_KEY' қосыңыз.")
    st.stop()

genai.configure(api_key=api_key)

SYSTEM_PROMPT = """
Сен — қазақ халқының салт-дәстүрлерін комикс форматында түсіндіретін ИИ-көмекшісің. 
Сенің аудиторияң — балалар мен дәстүрді жеңіл әрі қызықты түрде меңгергісі келетін ересектер.

### 💡 ЕРЕЖЕЛЕР:
1. Барлық сұраққа ТЕК ҚАЗАҚ ТІЛІНДЕ жауап бер.
2. Тіл жеңіл, достық, мейірімді, түсінікті болуы керек.
3. Әрқашан комикс форматында жауап бер.

### 🎬 ЖАУАП ФОРМАТЫ:
- **Кіріспе:** Сәлемдесу және дәстүрдің қысқаша атауы.
- **1-Кадр:**
  - 🎨 **Суретте:** (Визуалды көріністің сипаттамасы)
  - 💬 **Диалог:** (Кейіпкерлердің репликасы)
- **2-Кадр:**
  - 🎨 **Суретте:** ...
  - 💬 **Диалог:** ...
- **3-Кадр:**
  - 🎨 **Суретте:** ...
  - 💬 **Диалог:** ...
- **💡 Түсіндірме:** Дәстүрдің мағынасы мен тәрбиелік мәні (2-3 сөйлем).
- **❓ Сұрақ:** Пайдаланушыдан кері байланыс сұрау.
"""

model = genai.GenerativeModel(
    model_name="gemini-1.5-flash",
    system_instruction=SYSTEM_PROMPT
)

st.title("🎨 Қазақ дәстүрлері комиксте")
st.subheader("Салт-дәстүрлерді комикс арқылы оңай әрі қызықты үйрен!")

st.markdown("""
> Бұл қосымша балаларға да, ересектерге де қазақ халқының бай мәдениетін визуалды әрі қарапайым комикс форматында түсінуге көмектеседі.
""")

st.write("**Дайын сұрақтарды таңдаңыз немесе өзіңіз жазыңыз:**")
col1, col2, col3 = st.columns(3)

user_query = ""

if col1.button("👶 Тұсаукесер"):
    user_query = "Тұсаукесер дәстүрін комикс арқылы түсіндіріп берші."
if col2.button("🎁 Сүйінші"):
    user_query = "Сүйінші деген не?"
if col3.button("🏠 Шашу"):
    user_query = "Шашу дәстүрі туралы комикс жасап берші."

input_text = st.text_input("Қай дәстүр туралы білгіңіз келеді?", value=user_query, placeholder="Мысалы: Бата беру, Асар, Аластау...")

if st.button("✨ Комикс жасау", type="primary"):
    if input_text.strip() == "":
        st.warning("Өтініш, дәстүрдің атын жазыңыз!")
    else:
        with st.spinner("ИИ комикс сценарийін құрастыруда... 🎨"):
            try:
                response = model.generate_content(input_text)
                st.success("Дайын!")
                
                st.markdown("---")
                st.markdown(response.text)
                st.markdown("---")
            except Exception as e:
                st.error(f"Қателік орын алды: {e}")
